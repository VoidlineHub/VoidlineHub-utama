VoidlineBypass.HookDetection = function()

    for i, v in getgc(true) do
        if type(v) == "table" then
            local detected = rawget(v, "Detected")
            local kill = rawget(v, "Kill")
            local variables = rawget(v, "Variables")
            local process = rawget(v, "Process")
            
            if type(detected) == "function" and not detectedFunc then
                detectedFunc = detected
                if VoidlineBypass.Settings.DebugMode then
                    warn("[Voidline] Found Detected function")
                end
            end
            
            if variables and process and type(kill) == "function" and not killFunc then
                killFunc = kill
                if VoidlineBypass.Settings.DebugMode then
                    warn("[Voidline] Found Kill function")
                end
            end
        end
    end
    
    -- Hook detected function
    if detectedFunc then
        VoidlineBypass.OriginalFunctions.Detected = detectedFunc
        local hook; hook = hookfunction(detectedFunc, function(method, info)
            if VoidlineBypass.Settings.DebugMode then
                warn("[Voidline] Adonis Detection Blocked")
                warn("  Method: " .. tostring(method))
                warn("  Info: " .. tostring(info))
            end
            return true  -- jangan di utak atik
        end)
        table.insert(VoidlineBypass.HookedFunctions, hook)
    end
    
    -- modar-in hookfunction
    if killFunc then
        VoidlineBypass.OriginalFunctions.Kill = killFunc
        local hook; hook = hookfunction(killFunc, function(info)
            if VoidlineBypass.Settings.DebugMode then
                warn("[Voidline] Adonis Kill Blocked: " .. tostring(info))
            end
            -- sabar jing capek 
        end)
        table.insert(VoidlineBypass.HookedFunctions, hook)
    end
    
    return detectedFunc ~= nil or killFunc ~= nil
end

VoidlineBypass.BypassMetatableHooks = function()
    -- Check for proxy objects
    local proxies = {}
    
    for i, v in getgc(true) do
        if type(v) == "userdata" then
            local mt = getmetatable(v)
            if mt and rawget(mt, "__metatable") == "The metatable is locked" then
                table.insert(proxies, {object = v, metatable = mt})
            end
        end
    end
    
    -- Hook metatable methods
    for _, proxy in pairs(proxies) do
        local mt = proxy.metatable
        
        -- Replace dangerous metamethods
        local metamethods = {
            "__index", "__newindex", "__tostring", "__unm",
            "__add", "__sub", "__mul", "__div", "__mod",
            "__pow", "__len"
        }
        
        for _, mm in pairs(metamethods) do
            local original = rawget(mt, mm)
            if type(original) == "function" then
                rawset(mt, mm, function(...)
                    if VoidlineBypass.Settings.DebugMode then
                        warn("[Voidline] Metamethod blocked: " .. mm)
                    end
                    -- Return safe values
                    if mm == "__index" then return nil end
                    if mm == "__tostring" then return "Voidline Protected" end
                    if mm:find("__add") or mm:find("__sub") or mm:find("__mul") then
                        return 0
                    end
                    return nil
                end)
            end
        end
    end
    
    return #proxies > 0
end

--[[
    CLEANUP FUNCTION
    Restore original functions
]]
VoidlineBypass.Cleanup = function()
    if VoidlineBypass.Settings.DebugMode then
        warn("[Voidline] Cleaning up hooks...")
    end
    
    -- Disconnect all connections
    for _, connection in pairs(VoidlineBypass.HookedFunctions) do
        if type(connection) == "userdata" and connection.Disconnect then
            pcall(connection.Disconnect, connection)
        end
    end
    
    -- Restore original functions
    for funcName, original in pairs(VoidlineBypass.OriginalFunctions) do
        if original and type(original) == "function" then
            if hookfunction then
                pcall(hookfunction, _G[funcName], original)
            end
        end
    end
    
    VoidlineBypass.HookedFunctions = {}
    VoidlineBypass.OriginalFunctions = {}
    
    if VoidlineBypass.Settings.DebugMode then
        warn("[Voidline] Cleanup complete")
    end
end

-- UTILS
local function getHRP()
    local char = LP.Character or LP.CharacterAdded:Wait()
    return char:WaitForChild("HumanoidRootPart")
end

-- =========================================================
-- SAFE TWEEN (STAY AT HEIGHT -27 | SPEED 180)
-- =========================================================
local function safeTween(targetCF)
    local hrp = getHRP()
    local VOID_Y = -27
    local SPEED = 180 

    local downCF = CFrame.new(hrp.Position.X, VOID_Y, hrp.Position.Z)
    TweenService:Create(hrp, TweenInfo.new(0.5, Enum.EasingStyle.Sine), {CFrame = downCF}):Play()
    task.wait(0.55)

    local targetVoid = CFrame.new(targetCF.X, VOID_Y, targetCF.Z)
    local dist = (hrp.Position - targetVoid.Position).Magnitude
    local moveTween = TweenService:Create(hrp, TweenInfo.new(dist / SPEED, Enum.EasingStyle.Linear), {CFrame = targetVoid})
    moveTween:Play()
    moveTween.Completed:Wait()

    TweenService:Create(hrp, TweenInfo.new(0.5, Enum.EasingStyle.Sine), {CFrame = targetCF + Vector3.new(0, 2, 0)}):Play()
    task.wait(0.55)
end

-- =========================================================
-- PERSISTENT AUTOWALK (STAY AT SPEED 19)
-- =========================================================
local function autoWalk(targetPos)
    local char = LP.Character
    local hum = char and char:FindFirstChild("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end
    
    hum.WalkSpeed = 19
    
    repeat
        if not _G.AutoFarmBarista then break end
        hum:MoveTo(targetPos)
        task.wait(0.1)
    until (hrp.Position - targetPos).Magnitude < 3.8
end

-- =========================================================
-- SAFE CLICK (NO SPAM)
-- =========================================================
local function safeClick(prompt)
    if not prompt or not prompt.Enabled then return end
    
    task.wait(1.0) 
    fireproximityprompt(prompt)
    task.wait(0.5) 
end

-- =========================================================
-- REPAIR LOGIC
-- =========================================================
local function runRepairFlow()
    local missionUI = PlayerGui:FindFirstChild("BaristaMissionUI")
    local label = missionUI and missionUI:FindFirstChild("ObjectiveLabel", true)
    
    if label then
        task.wait(0.5) 
        local text = label.Text:lower()
        
        if text:find("anomaly") or text:find("strange") or text:find("person") then
            print("⚠️ Anomali Terdeteksi! Menunggu...")
            return 
        end

        if text:find("broke") or text:find("down") or text:find("fix") or text:find("restock") then
            isWorking = true
            print("🛠️ Machince Problem: OTW Supply...")
            
            safeTween(CFrame.new(POS_EXIT_BARISTA))
            safeTween(CFrame.new(POS_SUPPLY))
            
            BaristaEvent:FireServer("RestockSupplies")
            local prompt = supplyPart:FindFirstChild("SupplyPrompt")
            
            task.wait(1)
            fireproximityprompt(prompt)
            task.wait(6.5)
            
            safeTween(CFrame.new(POS_EXIT_BARISTA))
            isWorking = false
        end
    end
end

-- =========================
-- MINIGAME LOGIC
-- =========================
local function handleMinigame()
    local gui = PlayerGui:FindFirstChild("BaristaGUI")
    local mini = gui and gui:FindFirstChild("MinigameFrame")
    if not mini or not mini.Visible then return end

    isWorking = true
    pcall(function()
        local target = mini:WaitForChild("BackgroundBar"):WaitForChild("TargetZone")
        target.Size = UDim2.new(1.8, 0, 1.8, 0)
        target.Position = UDim2.new(-0.4, 0, -0.4, 0)
        target.BackgroundTransparency = 1
    end)

    while mini.Visible and _G.AutoFarmBarista do task.wait(0.5) end
    
    autoWalk(registerPart.Position)
    safeClick(registerPart:FindFirstChild("RegisterPrompt"))
    
    autoWalk(machinePart.Position)
    isWorking = false
end

-- MAIN CONTROLLER
-- =========================
local function StartBaristaGhost()
    if BaristaRunning then return end
    BaristaRunning = true

    task.spawn(function()
        TeamChangeRequest:FireServer("Barista", 11378976, 1, 0, "Detector")
        task.wait(2.5)
        
        while _G.AutoFarmBarista do
            local missionUI = PlayerGui:FindFirstChild("BaristaMissionUI")
            
            if not missionUI and not isWorking then
                safeTween(startPart.CFrame)
                safeClick(startPart:FindFirstChild("JobPrompt"))
                
                local retry = 0
                while not PlayerGui:FindFirstChild("BaristaMissionUI") and retry < 15 do
                    task.wait(1)
                    retry = retry + 1
                end
                
                if _G.AutoFarmBarista and PlayerGui:FindFirstChild("BaristaMissionUI") then
                    autoWalk(machinePart.Position)
                end

            elseif missionUI and not isWorking then
                local label = missionUI:FindFirstChild("ObjectiveLabel", true)
                if label and label.TextColor3 == Color3.fromRGB(255, 50, 50) then
                    runRepairFlow() -- Mangul fungsi repair lu
                end
            end

            if missionUI and not isWorking then
                local gui = PlayerGui:FindFirstChild("BaristaGUI")
                local mini = gui and gui:FindFirstChild("MinigameFrame")

                if mini and mini.Visible then
                    handleMinigame()
                else
                    local hrp = getHRP()
                    if (hrp.Position - machinePart.Position).Magnitude > 4 then
                        autoWalk(machinePart.Position)
                    else
                        local prompt = machinePart:FindFirstChild("MachinePrompt")
                        if prompt and prompt.Enabled then
                            safeClick(prompt)
                        end
                    end
                end
            end
            task.wait(0.5)
        end
        BaristaRunning = false
    end)
end
