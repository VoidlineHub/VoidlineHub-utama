local function FormatRupiah(num)
    local s = tostring(math.floor(num or 0))
    local result = s:reverse():gsub("(%d%d%d)", "%1."):reverse()
    return "Rp " .. (result:sub(1,1) == "." and result:sub(2) or result)
end

local function FormatTime(sec)
    return string.format("%02d:%02d:%02d", math.floor(sec/3600), math.floor((sec%3600)/60), sec%60)
end

-- [[ FORMAT TIME FIX ]] --
local function FormatTime(s)
    -- Pastikan detik tidak minus
    local secondsTotal = math.max(0, s)
    local hours = math.floor(secondsTotal / 3600)
    local minutes = math.floor((secondsTotal % 3600) / 60)
    local seconds = math.floor(secondsTotal % 60)
    return string.format("%02dh %02dm %02ds", hours, minutes, seconds)
end


local function SendWebhook(isTest, isDisconnect)
    local targetURL = _G.WebhookURL
    if not targetURL or targetURL == "" or not string.find(targetURL, "discord.com") then 
        return 
    end

    local runtimeSec = tick() - StartTime 
    local balanceVal = (MoneyValue and MoneyValue.Value) or 0
    
    local payload = {
        username = "VoidlineHub ⚡",
        embeds = {}
    }

    if isTest or isDisconnect then
        table.insert(payload.embeds, {
            title = isTest and "✅ WEBHOOK READY!" or "⚠️ SESSION ENDED!",
            description = isTest and "Webhook berhasil terhubung dengan **VoidlineHub** ⚡" or "Player telah keluar dari game/Server Disconnect.",
            color = isTest and 0x00FF88 or 0xFF0044,
            footer = { text = "Voidline System" },
            timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
        })
    else
        -- Update jobName biar sinkron sama fitur baru lu
        local jobName = "Idle"
        if _G.AutoFarm then jobName = "Mandalika Race"
        elseif _G.AutofarmMobil then jobName = "Shuttle Surakarta" -- Sesuaikan nama jobnya
        elseif _G.AutoFarmBandung then jobName = "Bandung Job"
        end

        table.insert(payload.embeds, {
            title = "📌 Farming Status Report",
            color = 0x1F8B4C,
            fields = {
                { name = "🧰 Active Job", value = jobName, inline = true },
                { name = "⏱ Runtime", value = FormatTime(runtimeSec), inline = true },
                { name = "💰 Earned", value = FormatRupiah(_G.TotalEarnedFix or 0), inline = true },
                { name = "💵 Balance", value = FormatRupiah(balanceVal), inline = true },
                { name = "📈 Est. Income", value = "Rp 30.000.000/Hr", inline = false }
            },
            footer = { text = "VoidlineHub System" },
            timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
        })
    end

    local requestFunc = syn and syn.request or http_request or request or (http and http.request)
    if requestFunc then
        pcall(function()
            requestFunc({
                Url = targetURL,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = HttpService:JSONEncode(payload)
            })
        end)
    end
end

local BlackScreenGui = nil
local function SetBlackScreen(state)
    if state then
        if not BlackScreenGui then
            BlackScreenGui = Instance.new("ScreenGui", CoreGui)
            BlackScreenGui.Name = "BilzzBlackScreen"
            BlackScreenGui.IgnoreGuiInset = true
            BlackScreenGui.DisplayOrder = -1

            Frame.Size = UDim2.new(1, 0, 1, 0)
            Frame.BackgroundColor3 = Color3.new(0, 0, 0)
            Frame.BorderSizePixel = 0

            local Label = Instance.new("TextLabel", Frame)
            Label.Size = UDim2.new(1, 0, 1, 0)
            Label.BackgroundTransparency = 1
            Label.Text = "AUTOFARM ON\nVoidLineHub"
            Label.TextColor3 = Color3.new(1, 1, 1) -- Tetap Putih
            Label.Font = Enum.Font.GothamBold
            Label.TextScaled = false
            Label.TextSize = 22

            -- Efek Kedap-Kedip (Blink)
            task.spawn(function()
                while BlackScreenGui do
                    -- Teks menghilang perlahan (Fade out)
                    local tweenOut = TweenService:Create(Label, TweenInfo.new(0.8), {TextTransparency = 1})

                    tweenIn.Completed:Wait()

                end
            end)
        end
    else
        if BlackScreenGui then
            BlackScreenGui:Destroy()
            BlackScreenGui = nil
        end
    end
end

-- [[ MOVEMENT ENGINE ]] --
local function flyTo(targetPos, moveSpeed, stopDistance)
    local v = getVehicle()
    if not v then return end
    local root = v:FindFirstChild("DriveSeat") or v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart", true)
    if not root then return end
    
    while (root.Position - targetPos).Magnitude > stopDistance and _G.AutoFarm do
        RunService.Heartbeat:Wait()
        local direction = (targetPos - root.Position).Unit
        root.AssemblyLinearVelocity = direction * moveSpeed
        root.CFrame = CFrame.new(root.Position, Vector3.new(targetPos.X, root.Position.Y, targetPos.Z))
        root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    end
end

local function startFarm()
    while _G.AutoRace do
        if not getVehicle() then 
            warn("DUDUK DULU!") 
            task.wait(2)
        else
            -- STEP 1: Ke Start
            moveTo(StartHitbox, StartSpeed, 2)
            
            -- CATAT WAKTU START
            local startTime = tick()
            
            -- STEP 2: Nunggu 30 Detik (Gantiin Sinyal Server)
            raceStarted = false
            
            repeat 
                -- Logika jaga posisi (Tween Lu yang Mulus)
                local v = getVehicle()
                if v then
                    local root = v:FindFirstChild("DriveSeat") or v.PrimaryPart
                    if root and (root.Position - StartHitbox).Magnitude > 5 then
                        root.CFrame = CFrame.new(StartHitbox, StartHitbox + Vector3.new(1,0,0))
                    end
                end
                task.wait(0.1) 
            until (tick() - startTime) >= 30 or not _G.AutoRace

            -- STEP 3: Balapan
            task.wait(0.1) -- Delay super tipis biar aman

            for _, point in ipairs(Route) do
                if not _G.AutoRace then break end
                -- stopDistance diubah ke 7 biar checkpoint gak gampang kelewat
                moveTo(point.pos + Vector3.new(0, Height, 0), Speed, 7)
                
                -- Jeda tipis banget biar server sempet registrasi koordinat CP
                task.wait(0.05) 
            end
            task.wait(3) -- Jeda 3 detik sebelum balik lagi ke start
        end
    end
end

_G.DCSent = false 

local function FinalExit()
    -- Cek apakah notif disconnect sudah dikirim atau URL kosong
    if _G.DCSent or not _G.WebhookURL or _G.WebhookURL == "" then return end
    _G.DCSent = true
    -- Kirim status Disconnect (isTest = false, isDisconnect = true)
    pcall(function() SendWebhook(false, true) end) 
end

-- [[ 1. MAIN AUTOMATION LOOP (WEBHOOK) ]] --
task.spawn(function()
    local lastUserSent = tick()
    
    while true do
        task.wait(1)
        if _G.DCSent then break end
        
        -- Cek apakah Webhook ON dan salah satu Farm lagi jalan
        local isFarming = _G.AutoFarm or _G.AutofarmMobil or _G.AutoFarmBandung
        
        if _G.WebhookEnabled and _G.WebhookURL ~= "" and isFarming then
            -- Kirim laporan setiap 50 detik
            if (tick() - lastUserSent) >= 50 then
                SendWebhook(false, false) -- Laporan rutin
                lastUserSent = tick()
            end
        else
            -- Reset timer kalau lagi gak farming biar gak "numpuk" pas diaktifin
            lastUserSent = tick() 
        end
    end
end)

-- [[ 3. MULTI-LOGIC (STEALTH & VEHICLE BOOST) ]] --
   -- --- A. Stealth Mode Logic ---
    -- 1. TARUH DI LUAR (DI ATAS HEARTBEAT)
local lastBoostTime = 0 

-- 2. BARU MASUK KE HEARTBEAT
RunService.Heartbeat:Connect(function()
    local char = LP.Character
    if not char then return end

    if _G.HideNameEnabled then
        local head = char:FindFirstChild("Head")
        if head then 
            for _, v in pairs(head:GetChildren()) do 
                if v:IsA("BillboardGui") then v.Enabled = false end 
            end 
        end
    end
    
    if _G.HideCharEnabled then
        for _, v in pairs(char:GetDescendants()) do 
            if v:IsA("BasePart") or v:IsA("Decal") then v.Transparency = 1 end 
        end
    end

    -- --- B. Vehicle Boost Logic ---

    -- (Logika Stealth kamu tetap di sini)

    -- 3. LOGIKA BOOST YANG SUDAH DI-FIX
    local hum = char:FindFirstChildOfClass("Humanoid")
    local seat = hum and hum.SeatPart
    
    if _G.BoostEnabled and seat then
        local vehicle = seat:FindFirstAncestorOfClass("Model")
        local root = (vehicle and vehicle.PrimaryPart) or seat -- Pakai seat kalau gak ada PrimaryPart
        
        if root then
            -- Cek interval
            if tick() - lastBoostTime >= (_G.Interval or 0.5) then
                local currentVel = root.AssemblyLinearVelocity
                local boostDirection = -root.CFrame.LookVector

                if _G.StraightBoost then
                    boostDirection = Vector3.new(boostDirection.X, 0, boostDirection.Z).Unit
                end
                
                -- PERBAIKAN: Gunakan 500 sebagai default limit, jangan 200 (karena 200 terlalu pelan)
                if currentVel.Magnitude < (_G.Speed or 300) then
                    root.AssemblyLinearVelocity = currentVel + (boostDirection * (_G.Increment or 50))
                    lastBoostTime = tick() -- Sekarang ini bakal kesimpan karena variabelnya ada di LUAR
                end
            end
        end
    end
end)

                    end

            end

        })

    Callback = function(v)
        _G.HideCharEnabled = v
        
        -- Logic instant buat balikin karakter pas toggle OFF
        if not v then
            local char = LP.Character
            if char then
                for _, part in pairs(char:GetDescendants()) do 
                    if part:IsA("BasePart") or part:IsA("Decal") then 
                        -- Jangan buat HumanoidRootPart jadi kelihatan (biasanya emang transparan)
                        if part.Name ~= "HumanoidRootPart" then
                            part.Transparency = 0 
                        end
                    end 
                end
            end
        end

        WindUI:Notify({
            Title = "Stealth Mode",
            Content = "Hide Character: " .. (v and "Enabled" or "Disabled"),
            Duration = 2
        })
    end
